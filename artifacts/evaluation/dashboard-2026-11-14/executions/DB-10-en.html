SELECT 
    DATE_TRUNC('month', wt.date) AS month,
    SUM(wt.sessions) AS sessions,
    SUM(wt.page_views) AS page_views
  FROM commerce.web_traffic AS wt
  WHERE EXTRACT(YEAR FROM wt.date) = $1
  GROUP BY 1
  ORDER BY 1 DESC
  LIMIT 200
